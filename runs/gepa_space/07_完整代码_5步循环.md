class SingleMutationSampling(SamplingStrategy):

# 绝对路径：src/gepa/gepa_utils.py:107-134（完整）
def select_program_candidate_from_pareto_front(pareto_front_programs, scores, rng) -> int:
    # 1. 去掉被支配的候选
    new_program_at_pareto_front_valset = remove_dominated_programs(pareto_front_programs, scores)
    # 2. 统计每个候选在多少个 testcase 的前沿里出现（freq）
    program_frequency_in_validation_pareto_front = {}
    for testcase_pareto_front in new_program_at_pareto_front_valset.values():
        for prog_idx in testcase_pareto_front:
            program_frequency_in_validation_pareto_front[prog_idx] = \
                program_frequency_in_validation_pareto_front.get(prog_idx, 0) + 1
    # 3. 按 freq 展开（freq=3 的出现 3 次 → 选中概率 3 倍）
    sampling_list = [prog_idx for prog_idx, freq in program_frequency_in_validation_pareto_front.items()
                     for _ in range(freq)]
    # 4. 有放回随机选一个索引（不移除）
    return rng.choice(sampling_list)
```

**关键结论**：选候选 = **按频率加权的有放回随机采样**——谁在更多子任务上领先（freq 高），谁被选中变异的机会越大。

# 绝对路径：src/gepa/proposer/reflective_mutation/reflective_mutation.py:286-714（propose 核心骨架）
def propose(self, state) -> list[CandidateProposal]:
    # Stage 1 ①选：采样 (父候选, minibatch) 任务
    tasks = self.sampling_strategy.sample_tasks(state, self.candidate_selector,
                                                 self.batch_sampler, self.trainset)

    # Stage 2 ②跑（父）：batch 评估父候选，capture_traces=True 采轨迹(ASI)
    unique_keys = {}  # 去重：同一候选不重复评估
    for task in tasks:
        unique_keys.setdefault(key, (task.parent_candidate, task.minibatch))
    parent_evals = self._batch_evaluate(items)   # → invoke_batch_evaluate(capture_traces=True)

    # Stage 3a ③反思前置：把评估结果打包成「反思数据集」
    for task in tasks:
        if not eval_curr.trajectories:      # 没轨迹(ASI)就跳过反思
            ... continue
        reflective_dataset = self.adapter.make_reflective_dataset(
            task.parent_candidate, eval_curr, predictor_names)

    # Stage 3b ③反思+④变异（合并）：LLM 读反思数据集 → 生成新文本
    jobs = [(parent_candidate, reflective_dataset, components) for ...]
    batch_texts = self._propose_texts_batch_safe(jobs)   # → reflection_lm.reflect()

    # Stage 3c：构建子候选（变异 = 替换组件的文本）
    for task, texts in ...:
        new_candidate = task.parent_candidate.copy()
        for name, text in new_texts.items():
            new_candidate[name] = text        # 变异：只改该组件的文本，其余不动

    # Stage 4 ②跑（子）：batch 评估子候选
    child_evals = self._batch_evaluate(child_items)

    # Stage 5：打包 proposal（带父/子的评估对比），返回给 engine 判收
    proposal = CandidateProposal(
        candidate=new_candidate,
        parent_program_ids=[task.parent_idx],
        eval_before=SubsampleEvaluation(scores=父候选分数, trajectories=父候选轨迹),
        eval_after =SubsampleEvaluation(scores=子候选分数, trajectories=子候选轨迹),
        tag="reflective_mutation",
    )
    return proposals
```

**关键结论**：`propose()` 只「生成」候选（①②③④），不「判收」。它把父/子的评估对比（eval_before/eval_after）打包成 proposal，交给 engine 的 `_run_reflective_batch` 判收。

## 五、一句话总结（5 步 → 代码的完整闭环）

```
主循环 engine.run()          while not 预算耗尽:
  ├─ propose()               = ①②③④（选/跑/反思/变异）—— 只生成，不判收
  │    └─ 选: sample_tasks → select_candidate_idx → rng.choice(加权采样)
  │    └─ 跑: invoke_batch_evaluate(capture_traces=True) 采轨迹
  │    └─ 反思: make_reflective_dataset 打包 side_info
  │    └─ 变异: reflection_lm.reflect() 读反思数据集 → 生成 new_texts
  └─ _run_reflective_batch() = ⑤收
       └─ acceptance_criterion.should_accept()  严格改进才收
       └─ update_state_with_new_program()       append 进候选池（只增不减）
       └─ 帕累托前沿: {program_idx} 替换旧冠军（无放回淘汰）
```

**两条数据流**：`score` 走「判收」（⑤），`side_info/trajectories` 走「反思」（③）。候选池「有放回只增不减」，帕累托前沿「无放回替换淘汰」。
