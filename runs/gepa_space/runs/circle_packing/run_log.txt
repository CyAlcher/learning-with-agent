def fitness_function(vars, n):
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Boundary constraints
    boundary_violations = np.sum(np.minimum(x - r, 1 - x - r, y - r, 1 - y - r))
    
    # Overlap constraints
    dist_matrix = cdist(np.column_stack((x, y)), np.column_stack((x, y)))
    np.fill_diagonal(dist_matrix, np.inf)
    overlap_violations = np.sum(np.maximum(0, (r[:, None] + r[None, :]) - dist_matrix))
    
    # Fitness is negative sum of radii with penalties
    return -np.sum(r) + 1e4 * boundary_violations + 1e4 * overlap_violations
